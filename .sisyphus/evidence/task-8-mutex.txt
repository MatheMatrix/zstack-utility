# Task 8: Pytest CLI Extension - Mutual Exclusion Validation Evidence

## Test 1: --ssh-host and --vm-deploy mutual exclusion
$ pytest tests/unit/test_mode_header.py --ssh-host=root:pass@192.168.1.100 --vm-deploy --target=192.168.1.100 2>&1


## Test 2: --vm-deploy requires --target
$ pytest tests/unit/test_mode_header.py --vm-deploy 2>&1


## Test 3: --ssh-host and --vm-deploy together error message check
$ pytest tests/unit/test_mode_header.py --ssh-host=x --vm-deploy 2>&1 | grep -i "exclusive"

ERROR: --ssh-host and --vm-deploy are mutually exclusive(cannot use both at the same time)

## Test 4: --vm-deploy without --target error message check
$ pytest tests/unit/test_mode_header.py --vm-deploy 2>&1 | grep -i "target"

ERROR: --vm-deploy requires --target to be specified (IP[:port] format)
